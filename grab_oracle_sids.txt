- name: Get Oracle SIDs from oratab
  ansible.builtin.shell: |
    awk -F: '!/^[[:space:]]*#/ && NF >= 2 {print $1}' /etc/oratab
  register: oratab_output
  changed_when: false

- name: Save Oracle SIDs
  ansible.builtin.set_fact:
    oracle_sids: "{{ oratab_output.stdout_lines }}"

oracle_sids:
  - ORCL
  - MGMTDB
  - +ASM

- name: Show Oracle SIDs
  ansible.builtin.debug:
    msg: "Oracle SID: {{ item }}"
  loop: "{{ oracle_sids }}"

---

- name: Check ordered inactive patches for each Oracle SID
  ansible.builtin.shell: |
    export ORACLE_SID="{{ oracle_sid }}"
    export ORAENV_ASK=NO

    . /usr/local/bin/oraenv >/dev/null 2>&1

    if [ -z "$ORACLE_HOME" ]; then
      echo "ERROR: ORACLE_HOME was not set for SID {{ oracle_sid }}" >&2
      exit 1
    fi

    "$ORACLE_HOME/OPatch/opatch" util listorderedinactivepatches \
      -oh "$ORACLE_HOME"
  args:
    executable: /bin/bash
  become: true
  become_user: >-
    {{
      'grid'
      if ('MGMT' in oracle_sid or 'ASM' in oracle_sid)
      else 'oracle'
    }}
  loop: "{{ oracle_sids }}"
  loop_control:
    loop_var: oracle_sid
    label: "{{ oracle_sid }}"
  register: opatch_results
  changed_when: false

- name: Show OPatch results
  ansible.builtin.debug:
    msg: |
      SID: {{ item.oracle_sid }}
      {{ item.stdout }}
  loop: "{{ opatch_results.results }}"
  loop_control:
    label: "{{ item.oracle_sid }}"

- name: Get Oracle SIDs from oratab
  ansible.builtin.shell: |
    awk -F: '!/^[[:space:]]*#/ && NF >= 2 {print $1}' /etc/oratab
  register: oratab_output
  changed_when: false

- name: Save Oracle SIDs
  ansible.builtin.set_fact:
    oracle_sids: "{{ oratab_output.stdout_lines }}"

oracle_sids:
  - ORCL
  - MGMTDB
  - +ASM

- name: Show Oracle SIDs
  ansible.builtin.debug:
    msg: "Oracle SID: {{ item }}"
  loop: "{{ oracle_sids }}"

---

- name: Get ORACLE_HOME for each Oracle SID
  ansible.builtin.shell: |
    export ORACLE_SID="{{ oracle_sid }}"
    export ORAENV_ASK=NO
    . oraenv >/dev/null 2>&1
    printf '%s\n' "$ORACLE_HOME"
  args:
    executable: /bin/bash
  become: true
  become_user: >-
    {{
      'grid'
      if ('MGMT' in oracle_sid or 'ASM' in oracle_sid)
      else 'oracle'
    }}
  loop: "{{ oracle_sids }}"
  loop_control:
    loop_var: oracle_sid
    label: "{{ oracle_sid }}"
  register: oracle_home_results
  changed_when: false

- name: Show discovered ORACLE_HOME values
  ansible.builtin.debug:
    msg: "{{ item.oracle_sid }} -> {{ item.stdout }}"
  loop: "{{ oracle_home_results.results }}"
  loop_control:
    label: "{{ item.oracle_sid }}"
